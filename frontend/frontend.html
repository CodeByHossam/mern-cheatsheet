import { createSlice } from '@reduxjs/toolkit';

const initialState = { value: 0 };

const counterSlice = createSlice({
  name: 'counter',
  initialState,
  reducers: {
    increment: (state) => { state.value += 1; },
    decrement: (state) => { state.value -= 1; },
    incrementByAmount: (state, action) => { state.value += action.payload; },
    reset: (state) => { state.value = 0; }
  }
});

export const { increment, decrement, incrementByAmount, reset } = counterSlice.actions;
export default counterSlice.reducer;

13. Frontend — RTK Query Base API + Endpoint Injection

install

npm i @reduxjs/toolkit react-redux

src/features/api/apiSlice.js

import { createApi, fetchBaseQuery } from '@reduxjs/toolkit/query/react';

const rawBaseQuery = fetchBaseQuery({
  baseUrl: '/api',
  credentials: 'include'
});

// retry once after a 401 by hitting /auth/refresh (rotates the cookie)
const baseQueryWithReauth = async (args, api, extraOptions) => {
  const result = await rawBaseQuery(args, api, extraOptions);
  if (result?.error?.status === 401) {
    const refresh = await rawBaseQuery('/auth/refresh', api, extraOptions);
    if (refresh?.data) {
      return rawBaseQuery(args, api, extraOptions);
    }
    api.dispatch({ type: 'auth/logout' });
  }
  return result;
};

export const apiSlice = createApi({
  reducerPath: 'api',
  baseQuery: baseQueryWithReauth,
  tagTypes: ['User', 'Auth'],
  endpoints: () => ({})
});
Note: Auth is cookie-based, so no prepareHeaders is needed — the browser sends the httpOnly cookie automatically. If you switch to a Bearer-token approach instead, re-add prepareHeaders reading from getState().auth or localStorage. Base URL: relative /api is proxied by Vite in dev (Section 10) and Nginx in prod (Section 22) — VITE_API_URL is only needed if you call the API cross-origin.

import { apiSlice } from './apiSlice';

export const usersApiSlice = apiSlice.injectEndpoints({
  endpoints: (builder) => ({
    getUsers: builder.query({
      query: () => '/users',
      // unwrap { success, data, message } so components read items directly
      transformResponse: (res) => res.data,
      providesTags: ['User']
    }),
    getUserById: builder.query({
      query: (id) => `/users/${id}`,
      transformResponse: (res) => res.data,
      providesTags: (r, e, id) => [{ type: 'User', id }]
    }),
    updateUser: builder.mutation({
      query: ({ id, ...body }) => ({
        url: `/users/${id}`,
        method: 'PUT',
        body
      }),
      invalidatesTags: (r, e, { id }) => [{ type: 'User', id }, 'User']
    })
  })
});

export const {
  useGetUsersQuery,
  useGetUserByIdQuery,
  useUpdateUserMutation
} = usersApiSlice;

import { createSlice } from '@reduxjs/toolkit';

const initialUser = JSON.parse(localStorage.getItem('user')) || null;

const authSlice = createSlice({
  name: 'auth',
  initialState: { user: initialUser },
  reducers: {
    setCredentials: (state, action) => {
      state.user = action.payload;
      localStorage.setItem('user', JSON.stringify(action.payload));
    },
    logout: (state) => {
      state.user = null;
      localStorage.removeItem('user');
    }
  }
});

export const { setCredentials, logout } = authSlice.actions;
export default authSlice.reducer;

  const { user } = useSelector((s) => s.auth);

  return <Outlet />;
}

src/components/ErrorBoundary.jsx

import { Component } from 'react';

export default class ErrorBoundary extends Component {
  constructor(props) {
    super(props);
    this.state = { hasError: false };
  }

  static getDerivedStateFromError() {
    return { hasError: true };
  }

  componentDidCatch(error, info) {
    console.error(error, info);
  }

  render() {
    if (this.state.hasError) {
      return (
        <div className="card m-8 text-center">
          <p className="font-bold">Something went wrong.</p>
        </div>
      );
    }
    return this.props.children;
  }
}

pages/Login.jsx — react-hook-form + zod + toast

import { useForm } from 'react-hook-form';
import { zodResolver } from '@hookform/resolvers/zod';
import { z } from 'zod';
import { useNavigate, useLocation } from 'react-router-dom';
import { useDispatch } from 'react-redux';
import toast from 'react-hot-toast';
import { useLoginMutation } from '../features/api/authApiSlice';
import { setCredentials } from '../features/auth/authSlice';

const schema = z.object({
  email: z.string().email('Invalid email'),
  password: z.string().min(6, 'Min 6 characters')
});

export default function Login() {
  const dispatch = useDispatch();
  const navigate = useNavigate();
  const location = useLocation();
  const [login, { isLoading }] = useLoginMutation();

  const { register, handleSubmit, formState: { errors } } = useForm({
    resolver: zodResolver(schema)
  });

  const onLogin = handleSubmit(async (values) => {
    try {
      const res = await login(values).unwrap(); // { success, data, message }
      dispatch(setCredentials(res.data));
      toast.success(res.message);
      navigate(location.state?.from?.pathname || '/dashboard', { replace: true });
    } catch (err) {
      toast.error(err?.data?.message || 'Login failed');
    }
  });

  return (
    <form onSubmit={onLogin} className="card max-w-sm mx-auto mt-10 space-y-4">
      <input className="input" placeholder="Email" {...register('email')} />
      {errors.email && <p className="text-red-600 text-sm">{errors.email.message}</p>}
      <input className="input" type="password" placeholder="Password" {...register('password')} />
      {errors.password && <p className="text-red-600 text-sm">{errors.password.message}</p>}
      <button className="btn-primary w-full" type="submit" disabled={isLoading}>
        {isLoading ? 'Signing in...' : 'Login'}
      </button>
    </form>
  );
}

15. WebSockets — Backend (Node.js) + Frontend (Vite + React)

REST is request/response — the client must poll. A WebSocket keeps one persistent connection open so the server can push (chat, live notifications, appointment updates) the moment something happens in MongoDB. Socket.IO wraps the WebSocket protocol with rooms, reconnection, and an auth handshake, and ships with a matching client for the Vite app.

Layer Package Role
Node backend socket.io Attaches to the same http.Server as Express — one port for API + realtime
Vite frontend socket.io-client Opens the socket; proxied by Vite in dev and Nginx in prod

install

# server
npm i socket.io

# client
npm i socket.io-client

server/socket.js — Socket.IO attached to Express (CommonJS)

const { Server } = require('socket.io');
const jwt = require('jsonwebtoken');
const env = require('./config/env');

let io; // one instance — REST controllers push through it

const initSocket = (httpServer) => {
  io = new Server(httpServer, {
    path: '/socket.io',
    cors: { origin: env.CLIENT_URL, credentials: true }
  });

  // handshake auth — the httpOnly token cookie is sent automatically
  io.use((socket, next) => {
    try {
      const header = socket.handshake.headers.cookie || '';
      const cookies = Object.fromEntries(
        header.split(';').map((c) => c.trim().split('='))
      );
      if (!cookies.token) return next(new Error('Not authorized'));
      socket.data.userId = jwt.verify(cookies.token, env.JWT_SECRET).id;
      next();
    } catch (err) {
      next(new Error('Invalid token'));
    }
  });

  io.on('connection', (socket) => {
    const userId = socket.data.userId;
    socket.join(`user:${userId}`); // private room per user

    // scope sockets to a resource (one appointment, one chat thread)
    socket.on('room:join', (room) => socket.join(room));
    socket.on('room:leave', (room) => socket.leave(room));

    socket.on('disconnect', () => {
      console.log('ws closed:', userId);
    });
  });

  return io;
};

// helpers reused by REST controllers (push right after a DB write)
const emitToUser = (userId, event, payload) =>
  io && io.to(`user:${userId}`).emit(event, payload);

const emitToRoom = (room, event, payload) =>
  io && io.to(room).emit(event, payload);

const getIO = () => io;

module.exports = { initSocket, emitToUser, emitToRoom, getIO };
Why a singleton? Controllers are required before the server boots, so they import emitToUser from this module instead of receiving io as a parameter — initSocket(server) fills the module-level io once at startup.

server.js — create the http server instead of app.listen()

const http = require('http');
const { initSocket, getIO } = require('./socket');

const start = async () => {
  try {
    await connectDB();

    const server = http.createServer(app); // Express still owns every /api route
    initSocket(server);                    // realtime shares the SAME port
    server.listen(env.PORT, () =>
      logger.info(`API + WS on ${env.PORT}`)
    );

    const shutdown = async (signal) => {
      logger.info(`${signal} received, shutting down...`);
      server.close(async () => {
        if (getIO()) getIO().close();
        await mongoose.connection.close();
        process.exit(0);
      });
      setTimeout(() => process.exit(1), 10000).unref();
    };
    process.on('SIGINT', () => shutdown('SIGINT'));
    process.on('SIGTERM', () => shutdown('SIGTERM'));
  } catch (err) {
    logger.error(err.message);
    process.exit(1);
  }
};

start();
Diff vs Section 4: only the listener changes — http.createServer(app) replaces app.listen(...). Middleware, routes, and app.use(notFound/errorHandler) stay untouched; upgrade requests for /socket.io never reach Express.

const Appointment = require('../models/Appointment');
const asyncHandler = require('../middleware/asyncHandler');
const { created } = require('../utils/apiResponse');
const { emitToUser } = require('../socket');

// @desc  create appointment   POST /api/appointments
exports.createAppointment = asyncHandler(async (req, res) => {
  const appointment = await Appointment.create({ ...req.body, user: req.user._id });

  emitToUser(req.user._id, 'appointment:created', {
    _id: appointment._id,
    date: appointment.date
  });

  created(res, appointment);
});

event contract (name it once, use it everywhere)

Event Direction Payload When
room:join / room:leave client → server 'appointment:<id>' User opens a detail page
appointment:created server → client { _id, date } POST /api/appointments succeeded
notification:new server → client { title, body } Any server-side event worth showing
presence:status server → client { userId, status } User connects / disconnects

import { io } from 'socket.io-client';

let socket = null;

// same-origin '/' → Vite proxy in dev, Nginx in prod; auth cookie rides along
export const connectSocket = () => {
  if (socket) return socket;
  socket = io('/', {
    path: '/socket.io',
    withCredentials: true,
    autoConnect: true,
    reconnection: true,
    reconnectionAttempts: 10
  });
  return socket;
};

export const disconnectSocket = () => {
  if (!socket) return;
  socket.disconnect();
  socket = null;
};

export const getSocket = () => socket;

client — src/hooks/useSocket.js

import { useEffect } from 'react';
import { connectSocket } from '../features/realtime/socket';

// subscribe to one event for the lifetime of the component
export default function useSocket(event, handler) {
  const socket = connectSocket();

  useEffect(() => {
    socket.on(event, handler);
    return () => socket.off(event, handler);
  }, [socket, event, handler]);

  return socket;
}

RTK Query — update the cache when the server pushes

import { apiSlice } from '../api/apiSlice';
import { connectSocket } from './socket';

export const appointmentsApiSlice = apiSlice.injectEndpoints({
  endpoints: (builder) => ({
    getAppointments: builder.query({
      query: () => '/appointments',
      providesTags: ['Appointment'],
      // keep the list live without refetching
      async onCacheEntryAdded(arg, { updateCachedData, cacheEntryRemoved }) {
        const socket = connectSocket();
        const onCreated = (appointment) =>
          updateCachedData((draft) => {
            draft.items.push(appointment);
          });

        socket.on('appointment:created', onCreated);
        await cacheEntryRemoved;   // cache evicted / component gone
        socket.off('appointment:created', onCreated);
      }
    })
  })
});

export const { useGetAppointmentsQuery } = appointmentsApiSlice;
Two-way sync: the mutation still invalidates providesTags (Section 13) for everyone already fetching — the socket listener only removes the gap between "row written to Mongo" and "next poll".

create — world, player, stars, score

let player, platforms, stars, bombs, score = 0, scoreText;

function create() {
  this.add.image(400, 300, 'sky');

  platforms = this.physics.add.staticGroup();
  platforms.create(400, 568, 'ground').setScale(2).refreshBody();
  platforms.create(600, 400, 'ground');
  platforms.create(50, 250, 'ground');
  platforms.create(750, 220, 'ground');

  player = this.physics.add.sprite(100, 450, 'dude');
  player.setBounce(0.2);
  player.setCollideWorldBounds(true);

  this.anims.create({ key: 'left', frames: this.anims.generateFrameNumbers('dude', { start: 0, end: 3 }), frameRate: 10, repeat: -1 });
  this.anims.create({ key: 'turn', frames: [{ key: 'dude', frame: 4 }], frameRate: 20 });
  this.anims.create({ key: 'right', frames: this.anims.generateFrameNumbers('dude', { start: 5, end: 8 }), frameRate: 10, repeat: -1 });

  this.physics.add.collider(player, platforms);

  stars = this.physics.add.group({ key: 'star', repeat: 11, setXY: { x: 12, y: 0, stepX: 70 } });
  stars.children.iterate((c) => c.setBounceY(Phaser.Math.FloatBetween(0.4, 0.8)));
  this.physics.add.collider(stars, platforms);
  this.physics.add.overlap(player, stars, collectStar, null, this);

  scoreText = this.add.text(16, 16, 'Score: 0', { fontSize: '32px', fill: '#fff' });
}

function collectStar(player, star) {
  star.disableBody(true, true);
  score += 10;
  scoreText.setText('Score: ' + score);
}

update — input + movement

function update() {
  const cursors = this.input.keyboard.createCursorKeys();
  if (cursors.left.isDown) {
    player.setVelocityX(-160);
    player.anims.play('left', true);
  } else if (cursors.right.isDown) {
    player.setVelocityX(160);
    player.anims.play('right', true);
  } else {
    player.setVelocityX(0);
    player.anims.play('turn');
  }
  if (cursors.up.isDown && player.body.touching.down) {
    player.setVelocityY(-330);
  }
}
