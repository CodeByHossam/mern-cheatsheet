18. Testing — Jest + Supertest (API) + Vitest + RTL (UI)

Two halves. The Express API is tested with Jest + Supertest against app.js (no listener); the Vite/React UI with Vitest + React Testing Library, which reuses the Vite transform pipeline (so JSX and ESM need no extra config). Prefer a single runner everywhere? Vitest runs the API tests too — see the end of part B.

A. API — Jest + Supertest

install (server)

npm i -D jest supertest

app / server split (required for tests)

// app.js — everything from server.js EXCEPT connectDB() and app.listen()
module.exports = app;
// server.js — now tiny
const app = require('./app');
const connectDB = require('./db');

const start = async () => {
  try {
    await connectDB();
    app.listen(env.PORT, () => console.log(`API on ${env.PORT}`));
  } catch (err) {
    console.error(err.message);
    process.exit(1);
  }
};
start();

tests/auth.test.js

const request = require('supertest');
const mongoose = require('mongoose');
const app = require('../app');
require('dotenv').config();

describe('Auth', () => {
  beforeAll(async () => {
    await mongoose.connect(process.env.MONGO_URI_TEST);
  });

  afterAll(async () => {
    await mongoose.connection.close();
  });

  it('registers a user', async () => {
    const res = await request(app)
      .post('/api/auth/register')
      .send({ name: 'Tester', email: 't@example.com', password: '123456' });
    expect(res.statusCode).toBe(201);
    expect(res.body.success).toBe(true);
  });

  it('rejects bad credentials', async () => {
    const res = await request(app)
      .post('/api/auth/login')
      .send({ email: 't@example.com', password: 'wrong' });
    expect(res.statusCode).toBe(401);
  });
});
// package.json scripts
"test": "jest --runInBand",
"test:watch": "jest --watch"
Note: Use a separate test DB via MONGO_URI_TEST in server/.env.test and add "jest": { "testEnvironment": "node" } to package.json. Run once with --runInBand to avoid parallel DB collisions.

B. UI — Vitest + React Testing Library

install (client)

npm i -D vitest jsdom @testing-library/react @testing-library/jest-dom @testing-library/user-event
npm i -D @vitest/coverage-v8        # coverage (v8 provider)
npm i -D @vitest/ui msw             # optional: visual test UI + network mocks

vite.config.js — the test block rides along with the Vite config

/// <reference types="vitest" />
import { defineConfig } from 'vite';
import react from '@vitejs/plugin-react';

export default defineConfig({
  plugins: [react()],
  test: {
    globals: true,                    // describe/it/expect without imports
    environment: 'jsdom',             // a DOM for React components
    setupFiles: './src/test/setup.js',
    css: false,                       // skip CSS processing inside tests
    coverage: { provider: 'v8', reporter: ['text', 'html'] }
  }
});
Alternative: keep the test config in vitest.config.js using defineConfig from vitest/config — same test key, but that file overrides vite.config.js. globals: true is optional: without it, import { describe, it, expect } from 'vitest' in every file (the tests below import explicitly, so both styles work).

src/test/setup.js — runs once before every test file

import '@testing-library/jest-dom/vitest';   // toBeInTheDocument(), toHaveTextContent(), …

// jsdom ships no matchMedia — stub it for components/hooks that query a breakpoint
Object.defineProperty(window, 'matchMedia', {
  writable: true,
  value: (query) => ({
    matches: false,
    media: query,
    onchange: null,
    addEventListener() {},
    removeEventListener() {},
    dispatchEvent: () => false
  })
});

src/test/renderWithProviders.jsx — one wrapper for store + router

import { render } from '@testing-library/react';
import { Provider } from 'react-redux';
import { MemoryRouter } from 'react-router-dom';
import { configureStore } from '@reduxjs/toolkit';
import counterReducer from '../features/counter/counterSlice';
import authReducer from '../features/auth/authSlice';
import { apiSlice } from '../features/api/apiSlice';

// a FRESH store per test — a shared store leaks state between tests
export const makeStore = () =>
  configureStore({
    reducer: {
      counter: counterReducer,
      auth: authReducer,
      [apiSlice.reducerPath]: apiSlice.reducer
    },
    middleware: (getDefault) => getDefault().concat(apiSlice.middleware)
  });

export function renderWithProviders(ui, { route = '/', store = makeStore() } = {}) {
  return {
    store,
    ...render(
      <Provider store={store}>
        <MemoryRouter initialEntries={[route]}>{ui}</MemoryRouter>
      </Provider>
    )
  };
}
Keep the test store in sync: if the app store grows (new slices, extra middleware), export a setupStore() factory from app/store.js (Section 12) and call it here too — then the tests can never drift from the real store.

1 — pure reducer test (no DOM, fastest)

// src/features/counter/counterSlice.test.js
import { describe, it, expect } from 'vitest';
import reducer, { increment, decrement, incrementByAmount, reset } from './counterSlice';

describe('counterSlice', () => {
  it('starts at 0', () => {
    expect(reducer(undefined, { type: 'init' })).toEqual({ value: 0 });
  });

  it('increments and decrements', () => {
    expect(reducer({ value: 1 }, increment()).value).toBe(2);
    expect(reducer({ value: 1 }, decrement()).value).toBe(0);
  });

  it('adds an amount, then resets', () => {
    const bumped = reducer({ value: 1 }, incrementByAmount(4));
    expect(bumped.value).toBe(5);
    expect(reducer(bumped, reset()).value).toBe(0);
  });
});

2 — component test (render + click, real store)

// src/features/counter/Counter.test.jsx  (the Section 12 component)
import { describe, it, expect } from 'vitest';
import { screen } from '@testing-library/react';
import userEvent from '@testing-library/user-event';
import { renderWithProviders } from '../../test/renderWithProviders';
import Counter from './Counter';

describe('<Counter />', () => {
  it('renders 0 and reacts to clicks', async () => {
    const user = userEvent.setup();
    renderWithProviders(<Counter />);

    expect(screen.getByText('0')).toBeInTheDocument();

    await user.click(screen.getByRole('button', { name: '+' }));
    await user.click(screen.getByRole('button', { name: '+' }));
    expect(screen.getByText('2')).toBeInTheDocument();

    await user.click(screen.getByRole('button', { name: 'Reset' }));
    expect(screen.getByText('0')).toBeInTheDocument();
  });
});
Query by role first: getByRole('button', { name: '+' }) asserts what a user (and a screen reader) can actually do. Use getByLabelText for form fields, findBy… for anything async, and waitFor only when a plain findBy… will not do.

3 — mock an API hook to test a page in isolation

// src/features/appointments/AppointmentsPage.test.jsx
import { describe, it, expect, vi } from 'vitest';
import { screen } from '@testing-library/react';
import { renderWithProviders } from '../../test/renderWithProviders';
import AppointmentsPage from './AppointmentsPage';
import * as api from './appointmentsApiSlice';   // the Section 15 slice

// swap the generated hook for a plain vi.fn() the test drives
vi.mock('./appointmentsApiSlice', () => ({ useGetAppointmentsQuery: vi.fn() }));

describe('<AppointmentsPage />', () => {
  it('renders the fetched appointments', () => {
    api.useGetAppointmentsQuery.mockReturnValue({
      data: { items: [{ _id: '1', patient: 'Sara' }] },
      isLoading: false,
      isError: false
    });

    renderWithProviders(<AppointmentsPage />);
    expect(screen.getByText('Sara')).toBeInTheDocument();
  });

  it('shows the error state', () => {
    api.useGetAppointmentsQuery.mockReturnValue({
      data: undefined,
      isLoading: false,
      isError: true
    });

    renderWithProviders(<AppointmentsPage />);
    expect(screen.getByText(/something went wrong/i)).toBeInTheDocument();
  });
});
Network-level tests: to exercise the real RTK Query path use msw — setupServer with http.get('/api/appointments', …) and beforeAll(() => server.listen({ onUnhandledRequest: 'error' })). Tests have no Vite dev proxy, so the API base must resolve to an absolute URL: set environmentOptions.jsdom.url and point VITE_API_URL at http://localhost:5000/api for the test run.

    listen 80;
    server_name yourdomain.com www.yourdomain.com;

    index index.html;

    location /api/ {
        proxy_pass http://127.0.0.1:5000/api/;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        try_files $uri $uri/ /index.html;
    }
}

23. SSL & Certificate Authority (CA)

Three routes: self-signed (dev only), CA-signed (your own CA → trust locally, e.g. for HTTPS in dev/internal networks), and Let's Encrypt (public prod). Create your own CA once, then sign as many server certs as you need.

create a local CA + sign a server certificate

# 1. CA key + self-signed CA certificate
openssl req -x509 -newkey rsa:2048 -nodes \
  -keyout ca-key.pem -out ca-cert.pem -days 3650 \
  -subj "/CN=My Local CA"

# 2. server key + Certificate Signing Request (CSR)
openssl req -newkey rsa:2048 -nodes \
  -keyout server-key.pem -out server.csr \
  -subj "/CN=localhost"

# 3. sign the CSR with the CA (SANs come from v3.ext)
openssl x509 -req -in server.csr \
  -CA ca-cert.pem -CAkey ca-key.pem -CAcreateserial \
  -out server-cert.pem -days 825 \
  -extfile v3.ext

v3.ext (subject alternative names required by browsers)

subjectAltName = DNS:localhost, IP:127.0.0.1
basicConstraints = CA:FALSE
keyUsage = digitalSignature, keyEncipherment
extendedKeyUsage = serverAuth

trust the CA (Ubuntu / browsers)

sudo cp ca-cert.pem /usr/local/share/ca-certificates/local-ca.crt
sudo update-ca-certificates
# also import ca-cert.pem into your BROWSER / mobile trust store
openssl verify -CAfile ca-cert.pem server-cert.pem   # expect: OK

quick self-signed cert (dev, no CA)

openssl req -x509 -newkey rsa:2048 -nodes \
  -keyout key.pem -out cert.pem -days 365 \
  -subj "/CN=localhost"

# inspect a cert / check expiry
openssl x509 -in cert.pem -text -noout
openssl x509 -enddate -noout -in cert.pem

HTTPS in Express (dev)

const https = require('https');
const fs = require('fs');

https
  .createServer(
    { key: fs.readFileSync('key.pem'), cert: fs.readFileSync('cert.pem') },
    app
  )
  .listen(8443);

public certificates — Let's Encrypt (reuse Section 22)

sudo apt install -y certbot python3-certbot-nginx
sudo certbot --nginx -d yourdomain.com -d www.yourdomain.com
sudo certbot renew --dry-run       # auto-renews via systemd timer
Best practice: always terminate TLS at a reverse proxy (Nginx/Caddy) in production — Express stays plain HTTP internally, and NEVER commit *.pem / private keys to git.
