4. Backend — Express Server Boilerplate

install

npm i express mongoose dotenv cookie-parser cors helmet hpp compression express-mongo-sanitize morgan swagger-ui-express

server.js

const express = require('express');
const mongoose = require('mongoose');
const cookieParser = require('cookie-parser');
const cors = require('cors');
const helmet = require('helmet');
const hpp = require('hpp');
const compression = require('compression');
const mongoSanitize = require('express-mongo-sanitize');
const morgan = require('morgan');
const swaggerUi = require('swagger-ui-express');
require('dotenv').config();

const connectDB = require('./db');
const env = require('./config/env');
const logger = require('./utils/logger');
const specs = require('./utils/swagger');
const apiLimiter = require('./middleware/rateLimitMiddleware');
const authRoutes = require('./routes/authRoutes');
const userRoutes = require('./routes/userRoutes');
const notFound = require('./middleware/notFoundMiddleware');
const errorHandler = require('./middleware/errorMiddleware');

const app = express();

// security & parsing
app.use(helmet());
app.use(cors({ origin: env.CLIENT_URL, credentials: true }));
app.use(express.json());
app.use(express.urlencoded({ extended: true }));
app.use(cookieParser());
app.use(mongoSanitize());
app.use(hpp());
app.use(compression());
app.use(morgan(env.NODE_ENV === 'production' ? 'combined' : 'dev'));

// rate limiter as middleware (applies to every /api/* request)
app.use('/api', apiLimiter);

// routes
app.use('/api/auth', authRoutes);
app.use('/api/users', userRoutes);
app.get('/api/health', (req, res) => res.json({ status: 'ok' }));

// api docs
app.use('/api-docs', swaggerUi.serve, swaggerUi.setup(specs));

// 404 + error
app.use(notFound);
app.use(errorHandler);

const start = async () => {
  try {
    await connectDB();
    const server = app.listen(env.PORT, () =>
      logger.info(`API on ${env.PORT}`)
    );

    // graceful shutdown
    const shutdown = async (signal) => {
      logger.info(`${signal} received, shutting down...`);
      server.close(async () => {
        await mongoose.connection.close();
        process.exit(0);
      });
      setTimeout(() => process.exit(1), 10000).unref();
    };
    process.on('SIGINT', () => shutdown('SIGINT'));
    process.on('SIGTERM', () => shutdown('SIGTERM'));
  } catch (err) {
    logger.error(err.message);
    process.exit(1);
  }
};

start();
Tip: For testability, extract the Express app to app.js (everything above start()) and module.exports = app;. Then server.js only imports it and runs start(). This is what the Jest tests in Section 17 use.

5. Backend — Common Middlewares

install

npm i jsonwebtoken express-rate-limit

middleware/asyncHandler.js

const asyncHandler = (fn) => (req, res, next) =>
  Promise.resolve(fn(req, res, next)).catch(next);

module.exports = asyncHandler;

middleware/authMiddleware.js

const jwt = require('jsonwebtoken');
const User = require('../models/User');
const env = require('../config/env');
const asyncHandler = require('./asyncHandler');

const protect = asyncHandler(async (req, res, next) => {
  const token = req.cookies.token || req.headers.authorization?.split(' ')[1];
  if (!token) {
    res.status(401);
    throw new Error('Not authorized');
  }
  const decoded = jwt.verify(token, env.JWT_SECRET);
  req.user = await User.findById(decoded.id).select('-password');
  if (!req.user) {
    res.status(401);
    throw new Error('User not found');
  }
  next();
});

module.exports = { protect };

middleware/adminMiddleware.js

const admin = (req, res, next) => {
  if (req.user && req.user.role === 'admin') return next();
  res.status(403);
  throw new Error('Admin only');
};

module.exports = { admin };

middleware/errorMiddleware.js

const errorHandler = (err, req, res, next) => {
  let status = res.statusCode === 200 ? 500 : res.statusCode;
  let message = err.message || 'Server error';

  if (err.name === 'CastError' && err.kind === 'ObjectId') {
    status = 404;
    message = 'Resource not found';
  }

  if (err.code === 11000) {
    status = 400;
    message = 'Duplicate value entered';
  }

  res.status(status).json({
    success: false,
    message,
    stack: process.env.NODE_ENV === 'production' ? undefined : err.stack
  });
};

module.exports = errorHandler;

middleware/notFoundMiddleware.js

const notFound = (req, res, next) => {
  res.status(404);
  next(new Error(`Not found - ${req.originalUrl}`));
};

module.exports = notFound;

middleware/rateLimitMiddleware.js

const rateLimit = require('express-rate-limit');

const apiLimiter = rateLimit({
  windowMs: 15 * 60 * 1000, // 15 min
  max: 300,                 // max requests per window per IP
  standardHeaders: true,    // return RateLimit-* headers
  legacyHeaders: false,     // disable X-RateLimit-* headers
  message: {
    success: false,
    message: 'Too many requests, please try again later'
  }
});

module.exports = apiLimiter;
Note: Mount it once in server.js with app.use('/api', apiLimiter) — it applies to every /api/* route as middleware, not inline.

every middleware used in server.js — at a glance

Middleware Source Purpose
helmet() helmet Security headers (CSP, HSTS, X-Frame-Options…)
cors() cors Allow CLIENT_URL origin + credentials cookie
express.json() express Parse JSON request bodies
express.urlencoded() express Parse form URL-encoded bodies
cookieParser() cookie-parser Parse the httpOnly auth cookies
mongoSanitize() express-mongo-sanitize Strip $ operators — blocks NoSQL injection
hpp() hpp Reject duplicate query params — blocks HTTP parameter pollution
compression() compression Gzip responses
morgan(...) morgan Request logging (combined in prod, dev in dev)
apiLimiter middleware/rateLimitMiddleware.js Rate limit /api/* per IP (above)
swaggerUi.serve swagger-ui-express Serve the /api-docs UI
notFound middleware/notFoundMiddleware.js 404 for unknown routes (above)
errorHandler middleware/errorMiddleware.js Central error response (above)
protect / admin route-level (above) Auth + role guard applied per route

third-party app-level middleware — config used in server.js

const helmet = require('helmet');
const cors = require('cors');
const mongoSanitize = require('express-mongo-sanitize');
const hpp = require('hpp');
const compression = require('compression');

app.use(helmet());                                // secure HTTP headers
app.use(cors({ origin: env.CLIENT_URL, credentials: true })); // single origin + cookies
app.use(express.json());                          // JSON bodies
app.use(express.urlencoded({ extended: true }));  // form bodies
app.use(cookieParser());                          // req.cookies.token etc.
app.use(mongoSanitize());                         // blocks $ne / $gt / $where injection
app.use(hpp());                                   // blocks ?id=1&id=2 pollution
app.use(compression());                           // gzip responses
app.use(morgan(env.NODE_ENV === 'production' ? 'combined' : 'dev'));
// optional strict CSP for production — tune directives to your app
app.use(
  helmet({
    contentSecurityPolicy: {
      directives: {
        defaultSrc: ["'self'"],
        scriptSrc: ["'self'"],
        styleSrc: ["'self'", "'unsafe-inline'"]
      }
    }
  })
);
Order matters: security + parsing middleware run before apiLimiter, routes, then notFound and finally errorHandler. Custom route guards (protect/admin) go per-route in routes/*.js.

install

models/User.js

const mongoose = require('mongoose');
const bcrypt = require('bcryptjs');

const userSchema = new mongoose.Schema(
  {
    name: { type: String, required: true, trim: true },
    email: { type: String, required: true, unique: true, lowercase: true },
    password: { type: String, required: true, minlength: 6 },
    role: { type: String, enum: ['user', 'admin'], default: 'user' }
  },
  { timestamps: true }
);

userSchema.pre('save', async function (next) {
  if (!this.isModified('password')) return next();
  const salt = await bcrypt.genSalt(10);
  this.password = await bcrypt.hash(this.password, salt);
  next();
});

userSchema.methods.matchPassword = function (entered) {
  return bcrypt.compare(entered, this.password);
};

module.exports = mongoose.model('User', userSchema);

utils/paginate.js

const paginate = (page = 1, limit = 10) => {
  const safePage = Math.max(parseInt(page, 10) || 1, 1);
  const safeLimit = Math.min(Math.max(parseInt(limit, 10) || 10, 1), 100);
  return { skip: (safePage - 1) * safeLimit, limit: safeLimit, page: safePage };
};

const paginatedResult = (items, count, page, limit) => ({
  items,
  pagination: { page, limit, total: count, pages: Math.ceil(count / limit) }
});

module.exports = { paginate, paginatedResult };

controllers/userController.js

const User = require('../models/User');
const asyncHandler = require('../middleware/asyncHandler');
const { ok, created } = require('../utils/apiResponse');
const { paginate, paginatedResult } = require('../utils/paginate');

// @desc  list users (paginated)   GET /api/users?page=1&limit=10
exports.getUsers = asyncHandler(async (req, res) => {
  const { skip, limit, page } = paginate(req.query.page, req.query.limit);
  const [items, count] = await Promise.all([
    User.find().select('-password').skip(skip).limit(limit).lean(),
    User.countDocuments()
  ]);
  ok(res, paginatedResult(items, count, page, limit));
});

// @desc  single user
exports.getUserById = asyncHandler(async (req, res) => {
  const user = await User.findById(req.params.id).select('-password').lean();
  if (!user) {
    res.status(404);
    throw new Error('User not found');
  }
  ok(res, user);
});

// @desc  create
exports.createUser = asyncHandler(async (req, res) => {
  const user = await User.create(req.body);
  created(res, { _id: user._id, name: user.name, email: user.email });
});

// @desc  update
exports.updateUser = asyncHandler(async (req, res) => {
  const user = await User.findByIdAndUpdate(req.params.id, req.body, {
    new: true,
    runValidators: true
  }).select('-password');
  if (!user) {
    res.status(404);
    throw new Error('User not found');
  }
  ok(res, user);
});

// @desc  delete
exports.deleteUser = asyncHandler(async (req, res) => {
  const user = await User.findByIdAndDelete(req.params.id);
  if (!user) {
    res.status(404);
    throw new Error('User not found');
  }
  ok(res, null, 'User removed');
});

controllers/authController.js

const User = require('../models/User');
const jwt = require('jsonwebtoken');
const env = require('../config/env');
const asyncHandler = require('../middleware/asyncHandler');
const { ok, created } = require('../utils/apiResponse');

const signToken = (id) =>
  jwt.sign({ id }, env.JWT_SECRET, { expiresIn: env.JWT_EXPIRES_IN });

const setCookie = (res, token) =>
  res.cookie('token', token, {
    httpOnly: true,
    secure: env.NODE_ENV === 'production',
    sameSite: 'lax',
    maxAge: 15 * 60 * 1000 // matches JWT_EXPIRES_IN
  });

// @desc  register   POST /api/auth/register
exports.register = asyncHandler(async (req, res) => {
  const { name, email, password } = req.body;
  const user = await User.create({ name, email, password });
  const token = signToken(user._id);
  setCookie(res, token);
  created(res, { _id: user._id, name, email, role: user.role }, 'Registered');
});

// @desc  login      POST /api/auth/login
exports.login = asyncHandler(async (req, res) => {
  const { email, password } = req.body;
  const user = await User.findOne({ email });
  if (!user || !(await user.matchPassword(password))) {
    res.status(401);
    throw new Error('Invalid credentials');
  }
  const token = signToken(user._id);
  setCookie(res, token);
  ok(res, { _id: user._id, name: user.name, email: user.email, role: user.role }, 'Logged in');
});

// @desc  logout     POST /api/auth/logout
exports.logout = asyncHandler(async (req, res) => {
  res.clearCookie('token');
  ok(res, null, 'Logged out');
});

// @desc  current user  GET /api/auth/me
exports.getMe = asyncHandler(async (req, res) => {
  const user = await User.findById(req.user._id).select('-password').lean();
  ok(res, user);
});

const { body, validationResult } = require('express-validator');

const registerRules = [
  body('name').trim().notEmpty().withMessage('Name is required'),
  body('email').isEmail().normalizeEmail().withMessage('Valid email required'),
  body('password')
    .isLength({ min: 6 })
    .withMessage('Password must be at least 6 characters')
];

const loginRules = [
  body('email').isEmail().normalizeEmail().withMessage('Valid email required'),
  body('password').notEmpty().withMessage('Password is required')
];

const validate = (req, res, next) => {
  const errors = validationResult(req);
  if (!errors.isEmpty()) {
    return res.status(400).json({
      success: false,
      message: errors.array().map((e) => e.msg).join(', ')
    });
  }
  next();
};

module.exports = { registerRules, loginRules, validate };
// use in authRoutes.js

9. Auth Deep-Dive — JWT Access + Refresh (httpOnly Cookies)

install

npm i jsonwebtoken cookie-parser
Token Lifetime Storage Purpose
Access (JWT) 15m httpOnly cookie token Authorizes /api/* requests
Refresh (JWT) 30d httpOnly cookie refreshToken Silently mints a new access token at POST /api/auth/refresh

Access tokens are short-lived so a stolen token expires fast; the refresh token silently renews the session. Both live in httpOnly cookies, so JavaScript can't read them (XSS-safe).

utils/token.js

const jwt = require('jsonwebtoken');
const env = require('../config/env');

const accessCookie = {
  httpOnly: true,
  secure: env.NODE_ENV === 'production',
  sameSite: 'lax',
  maxAge: 15 * 60 * 1000
};

const refreshCookie = {
  ...accessCookie,
  maxAge: 30 * 24 * 60 * 60 * 1000
};

const signAccess = (id) => jwt.sign({ id }, env.JWT_SECRET, { expiresIn: env.JWT_EXPIRES_IN });
const signRefresh = (id) => jwt.sign({ id }, env.REFRESH_SECRET, { expiresIn: env.REFRESH_EXPIRES_IN });

module.exports = { accessCookie, refreshCookie, signAccess, signRefresh };

controllers/tokenController.js — refresh + rotate

const jwt = require('jsonwebtoken');
const User = require('../models/User');
const env = require('../config/env');
const asyncHandler = require('../middleware/asyncHandler');
const { ok } = require('../utils/apiResponse');
const { accessCookie, refreshCookie, signAccess, signRefresh } = require('../utils/token');

// POST /api/auth/refresh
exports.refresh = asyncHandler(async (req, res) => {
  const refreshToken = req.cookies.refreshToken;
  if (!refreshToken) {
    res.status(401);
    throw new Error('No refresh token');
  }

  const decoded = jwt.verify(refreshToken, env.REFRESH_SECRET);
  const user = await User.findById(decoded.id).select('-password');
  if (!user) {
    res.status(401);
    throw new Error('User not found');
  }

  res.cookie('token', signAccess(user._id), accessCookie);
  res.cookie('refreshToken', signRefresh(user._id), refreshCookie);
  ok(res, { _id: user._id, name: user.name, email: user.email, role: user.role }, 'Refreshed');
});
Security notes: Mount router.post('/refresh', refresh) in authRoutes.js. The RTK Query baseQueryWithReauth (Section 13) calls this endpoint once on a 401. For rotation-logout, persist a hashed refresh token in Mongo and revoke it on logout; otherwise logout is client-side only.
